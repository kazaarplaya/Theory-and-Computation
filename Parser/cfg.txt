Program -> ClassDeclarationList

// Classes + Members
ClassDeclarationList -> ClassDeclaration | ClassDeclaration ClassDeclarationList
ClassDeclarationList -> "empty"

ClassDeclaration -> "class" Identifier "{" "MemberList "}" ";"

MemberList -> MemberDeclaration | MemberDeclaration MemberList

MemberDeclaration -> Visibility | Visibility VarDeclaration
MemberDeclaration -> Visibility FunctionDeclaration

Visibility -> "public"
Visibility -> "private"
Visibility -> "protected"

VarDeclaration -> Type Identifier ";"

Type -> "int"
Type -> "char"

FunctionDeclaration -> Identifier "(" FunctionParameters ")" Block

// Parameter
FunctionParameters -> Parameter ParameterListTail
FunctionParameters -> empty

Parameter -> Type Identifier

ParameterList Tail -> "," Parameter

// Body + Functions
Block -> "{" FunctionBody "}"

FunctionBody -> VarDeclaration;
FunctionBody -> Assignment
FunctionBody -> ReturnStatement 
FunctionBody -> IfStatement
FunctionBody -> WhileStatement
FunctionBody -> ForStatement
FunctionBody -> DoStatement

Assignment -> Type Identifier "=" Integer ";"

ReturnStatement -> "return" ";"
ReturnStatement -> "return" Integer ";"
ReturnStatement -> "return" Identifier ";"

IfStatement -> "if" Expression Block
IfStatement -> "if" Expression Block "else" Block

WhileStatement -> "while" "(" Expression ")" Block

// For Loop
ForStatement -> "for" "(" ForInitialisation ";" Expression ";" Assignment ")" Block 

ForInitialisation -> empty
ForInitialisation -> AssignmentWithNoSemi
ForInitialisation -> Assignment

AssignmentWithNoSemi -> Type Identifier "=" Integer

DoStatement -> "do" Block "while" "(" Condition ")" ";"

// Expressions, Terms, Factors
Expression -> Expression "+" Term
Expression -> Expression "-" Term
Expression -> Term

Term -> Term "*" Factor
Term -> Term "/" Factor
Term -> Term "%" Factor
Term -> Factor

Factor -> Identifier
Factor -> Integer
